 8def JER(ref, hyp, collar=0.0, skip_overlap=False, detailed=False):
 9    """Compute Jaccard Error Rate (JER).
10
11    JER is the average over all reference speakers of (1 - Jaccard similarity)
12    under an optimal 1-to-1 mapping between reference and hypothesis speakers:
13
14        JER_r = 1 - |r \\cap h| / |r \\cup h|
15        JER = (1 / |S_ref|) * \\sum_{r \\in S_ref} JER_r
16
17    Unmapped reference speakers have JER_r = 1.0.
18
19    Args:
20        ref: a list of tuples for the ground truth, where each tuple is
21            (speaker, start, end) of type (string, float, float)
22        hyp: a list of tuples for the diarization result hypothesis, same type
23            as `ref`
24        collar: float, no-score zone (in seconds) around each reference
25            speaker boundary ([boundary - collar, boundary + collar])
26        skip_overlap: bool, if True, exclude regions where more than one
27            reference speaker is active
28        detailed: bool, if True, return a dict containing overall JER,
29            per-reference-speaker JER, and the optimal speaker_mapping
30
31    Returns:
32        If `detailed=False`, a float number for the Jaccard Error Rate.
33        If `detailed=True`, a dict with keys:
34            - "jer": float, Jaccard Error Rate
35            - "speaker_jer": dict mapping each ref speaker -> speaker JER
36            - "speaker_mapping": dict mapping matched ref -> hyp speaker
37    """
38    if not isinstance(detailed, bool):
39        raise TypeError("detailed must be a boolean.")
40    ref, hyp = der._preprocess_segments(
41        ref, hyp, collar=collar, skip_overlap=skip_overlap)
42
43    ref_index = der.build_speaker_index(ref)
44    hyp_index = der.build_speaker_index(hyp)
45
46    if not ref_index:
47        if detailed:
48            return {
49                "jer": 0.0,
50                "speaker_jer": {},
51                "speaker_mapping": {},
52            }
53        return 0.0
54
55    ref_durations = np.zeros(len(ref_index))
56    for spk, start, end in ref:
57        ref_durations[ref_index[spk]] += end - start
58
59    hyp_durations = np.zeros(len(hyp_index))
60    for spk, start, end in hyp:
61        hyp_durations[hyp_index[spk]] += end - start
62
63    intersection_matrix = der.build_cost_matrix(ref, hyp)
64    union_matrix = (
65        ref_durations[:, None] + hyp_durations[None, :] - intersection_matrix
66    )
67    jaccard_matrix = np.divide(
68        intersection_matrix,
69        union_matrix,
70        out=np.zeros_like(intersection_matrix),
71        where=union_matrix > 0.0,
72    )
73
74    row_index, col_index = optimize.linear_sum_assignment(-jaccard_matrix)
75
76    inv_ref = {i: spk for spk, i in ref_index.items()}
77    inv_hyp = {j: spk for spk, j in hyp_index.items()}
78    speaker_jer = {spk: 1.0 for spk in ref_index}
79    speaker_mapping = {}
80
81    for r, c in zip(row_index, col_index):
82        ref_spk = inv_ref[r]
83        speaker_jer[ref_spk] = float(1.0 - jaccard_matrix[r, c])
84        if intersection_matrix[r, c] > 0.0:
85            speaker_mapping[ref_spk] = inv_hyp[c]
86
87    jer_val = float(sum(speaker_jer.values()) / len(ref_index))
88
89    if detailed:
90        return {
91            "jer": jer_val,
92            "speaker_jer": speaker_jer,
93            "speaker_mapping": speaker_mapping,
94        }
95    return jer_val

139def WDER(ref, hyp, detailed=False):
140    """Compute Word Diarization Error Rate (WDER).
141
142    WDER measures the fraction of aligned words (correct ASR words plus
143    substituted ASR words after Levenshtein alignment between reference and
144    hypothesis transcripts) that are assigned to the wrong speaker after
145    optimal 1-to-1 speaker mapping (Shafey et al., Interspeech 2019).
146
147    Args:
148        ref: a list of (speaker, word) tuples of type (str, str) for the
149            ground truth. Each tuple's text may be a single word or
150            whitespace-separated words spoken by that speaker.
151        hyp: a list of (speaker, word) tuples of type (str, str) for the
152            hypothesis, same format as `ref`.
153        detailed: bool, if True, return a dict containing WDER, word counts
154            (wder_sub, wder_correct, wder_total), WER breakdown, and the
155            optimal speaker_mapping.
156
157    Returns:
158        If `detailed=False`, a float number for the Word Diarization Error
159        Rate.
160        If `detailed=True`, a dict with keys:
161            - "wder": float, Word Diarization Error Rate
162            - "wder_sub": int, aligned words with wrong speaker label
163            - "wder_correct": int, aligned words with correct speaker label
164            - "wder_total": int, total aligned words (wer_correct + wer_sub)
165            - "wer": float, Word Error Rate of the transcript
166            - "wer_insert": int, inserted words
167            - "wer_delete": int, deleted words
168            - "wer_sub": int, substituted words
169            - "wer_correct": int, correct words
170            - "wer_total": int, total reference words
171            - "speaker_mapping": dict mapping matched ref -> hyp speaker
172    """
173    if not isinstance(detailed, bool):
174        raise TypeError("detailed must be a boolean.")
175    check_word_input(ref)
176    check_word_input(hyp)
177
178    ref_expanded = _expand_word_tuples(ref)
179    hyp_expanded = _expand_word_tuples(hyp)
180
181    ref_spks = [spk for spk, _ in ref_expanded]
182    ref_words = [w for _, w in ref_expanded]
183    hyp_spks = [spk for spk, _ in hyp_expanded]
184    hyp_words = [w for _, w in hyp_expanded]
185
186    _, align = levenshtein_with_edits(ref_words, hyp_words)
187
188    wer_insert = 0
189    wer_delete = 0
190    wer_sub = 0
191    wer_correct = 0
192    aligned_spk_pairs = []
193
194    for i, j in align:
195        if i == -1:
196            wer_insert += 1
197        elif j == -1:
198            wer_delete += 1
199        else:
200            if ref_words[i] == hyp_words[j]:
201                wer_correct += 1
202            else:
203                wer_sub += 1
204            aligned_spk_pairs.append((ref_spks[i], hyp_spks[j]))
205
206    wer_total = len(ref_words)
207    wer_val = (
208        0.0 if wer_total == 0
209        else float((wer_insert + wer_delete + wer_sub) / wer_total)
210    )
211
212    wder_total = len(aligned_spk_pairs)
213    if wder_total == 0:
214        wder_correct = 0
215        wder_sub = 0
216        wder_val = 0.0
217        speaker_mapping = {}
218    else:
219        ref_index = der.build_speaker_index(ref_expanded)
220        hyp_index = der.build_speaker_index(hyp_expanded)
221        cost_matrix = np.zeros((len(ref_index), len(hyp_index)), dtype=int)
222        for r_spk, h_spk in aligned_spk_pairs:
223            cost_matrix[ref_index[r_spk], hyp_index[h_spk]] += 1
224
225        row_index, col_index = optimize.linear_sum_assignment(
226            cost_matrix, maximize=True)
227        wder_correct = int(cost_matrix[row_index, col_index].sum())
228        wder_sub = wder_total - wder_correct
229        wder_val = float(wder_sub / wder_total)
230
231        inv_ref = {i: spk for spk, i in ref_index.items()}
232        inv_hyp = {j: spk for spk, j in hyp_index.items()}
233        speaker_mapping = {
234            inv_ref[r]: inv_hyp[c]
235            for r, c in zip(row_index, col_index)
236            if cost_matrix[r, c] > 0
237        }
238
239    if detailed:
240        return {
241            "wder": wder_val,
242            "wder_sub": wder_sub,
243            "wder_correct": wder_correct,
244            "wder_total": wder_total,
245            "wer": wer_val,
246            "wer_insert": wer_insert,
247            "wer_delete": wer_delete,
248            "wer_sub": wer_sub,
249            "wer_correct": wer_correct,
250            "wer_total": wer_total,
251            "speaker_mapping": speaker_mapping,
252        }
253    return wder_val

451

462            per-recording results
463
464    Returns:
465        If `detailed=False`, a float number for the Corpus DER.
466        If `detailed=True`, a dict with keys:
467            - "der": float, Corpus DER
468            - "miss": float, total missed speech duration in seconds
469            - "false_alarm": float, total false alarm duration in seconds
470            - "confusion": float, total speaker confusion duration in seconds
471            - "ref_length": float, total scored reference length in seconds
472            - "per_recording": dict or list of per-recording detailed results

479    total_fa = 0.0
480    total_confusion = 0.0
481    total_ref_length = 0.0

486            ref, hyp, collar=collar, skip_overlap=skip_overlap, detailed=True)

10

13

15        rttm_source: path to an RTTM file (str or os.PathLike) or an iterable

17
18    Returns:
19        a dict mapping `recording_id` (str) to a list of
20        `(speaker, start, end)` tuples
21    """
22    if isinstance(rttm_source, (str, os.PathLike)):
23        with open(rttm_source, "r", encoding="utf-8") as f:
24            lines = f.readlines()
25    elif hasattr(rttm_source, "__iter__") and not isinstance(
26            rttm_source, (bytes, bytearray)):
27        lines = list(rttm_source)
28    else:
29        raise TypeError(
30            "rttm_source must be a file path or an iterable of lines.")
31
32    recordings = {}
33    for line in lines:
34        stripped = line.strip()
35        if not stripped or stripped.startswith(("#", ";")):
36            continue
37        parts = stripped.split()
38        if parts[0].upper() != "SPEAKER":
39            continue
40        if len(parts) < 8:
41            raise ValueError(f"Invalid RTTM SPEAKER line: {stripped}")
42        rec_id = parts[1]
43        start = float(parts[3])
44        duration = float(parts[4])
45        if not np.isfinite(start) or not np.isfinite(duration) or duration < 0:
46            raise ValueError(
47                f"Invalid start or duration in RTTM line: {stripped}")
48        end = start + duration
49        speaker = parts[7]
50        recordings.setdefault(rec_id, []).append((speaker, start, end))
51
52    return recordings
