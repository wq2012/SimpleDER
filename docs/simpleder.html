439def JER(ref, hyp, collar=0.0, skip_overlap=False, detailed=False):
440    """Compute Jaccard Error Rate (JER).
441
442    JER is the average over all reference speakers of (1 - Jaccard similarity)
443    under an optimal 1-to-1 mapping between reference and hypothesis speakers:
444
445        JER_r = 1 - |r \\cap h| / |r \\cup h|
446        JER = (1 / |S_ref|) * \\sum_{r \\in S_ref} JER_r
447
448    Unmapped reference speakers have JER_r = 1.0.
449
450    Args:
451        ref: a list of tuples for the ground truth, where each tuple is
452            (speaker, start, end) of type (string, float, float)
453        hyp: a list of tuples for the diarization result hypothesis, same type
454            as `ref`
455        collar: float, no-score zone (in seconds) around each reference
456            speaker boundary ([boundary - collar, boundary + collar])
457        skip_overlap: bool, if True, exclude regions where more than one
458            reference speaker is active
459        detailed: bool, if True, return a dict containing overall JER,
460            per-reference-speaker JER, and the optimal speaker_mapping
461
462    Returns:
463        If `detailed=False`, a float number for the Jaccard Error Rate.
464        If `detailed=True`, a dict with keys:
465            - "jer": float, Jaccard Error Rate
466            - "speaker_jer": dict mapping each ref speaker -> speaker JER
467            - "speaker_mapping": dict mapping matched ref -> hyp speaker
468    """
469    if not isinstance(detailed, bool):
470        raise TypeError("detailed must be a boolean.")
471    ref, hyp = _preprocess_segments(
472        ref, hyp, collar=collar, skip_overlap=skip_overlap)
473
474    ref_index = build_speaker_index(ref)
475    hyp_index = build_speaker_index(hyp)
476
477    if not ref_index:
478        if detailed:
479            return {
480                "jer": 0.0,
481                "speaker_jer": {},
482                "speaker_mapping": {},
483            }
484        return 0.0
485
486    ref_durations = np.zeros(len(ref_index))
487    for spk, start, end in ref:
488        ref_durations[ref_index[spk]] += end - start
489
490    hyp_durations = np.zeros(len(hyp_index))
491    for spk, start, end in hyp:
492        hyp_durations[hyp_index[spk]] += end - start
493
494    intersection_matrix = build_cost_matrix(ref, hyp)
495    union_matrix = (
496        ref_durations[:, None] + hyp_durations[None, :] - intersection_matrix
497    )
498    jaccard_matrix = np.divide(
499        intersection_matrix,
500        union_matrix,
501        out=np.zeros_like(intersection_matrix),
502        where=union_matrix > 0.0,
503    )
504
505    row_index, col_index = optimize.linear_sum_assignment(-jaccard_matrix)
506
507    inv_ref = {i: spk for spk, i in ref_index.items()}
508    inv_hyp = {j: spk for spk, j in hyp_index.items()}
509    speaker_jer = {spk: 1.0 for spk in ref_index}
510    speaker_mapping = {}
511
512    for r, c in zip(row_index, col_index):
513        ref_spk = inv_ref[r]
514        speaker_jer[ref_spk] = float(1.0 - jaccard_matrix[r, c])
515        if intersection_matrix[r, c] > 0.0:
516            speaker_mapping[ref_spk] = inv_hyp[c]
517
518    jer_val = float(sum(speaker_jer.values()) / len(ref_index))
519
520    if detailed:
521        return {
522            "jer": jer_val,
523            "speaker_jer": speaker_jer,
524            "speaker_mapping": speaker_mapping,
525        }
526    return jer_val

652def WDER(ref, hyp, detailed=False):
653    """Compute Word Diarization Error Rate (WDER).
654
655    WDER measures the fraction of aligned words (correct ASR words plus
656    substituted ASR words after Levenshtein alignment between reference and
657    hypothesis transcripts) that are assigned to the wrong speaker after
658    optimal 1-to-1 speaker mapping (Shafey et al., Interspeech 2019).
659
660    Args:
661        ref: a list of (speaker, word) tuples of type (str, str) for the
662            ground truth. Each tuple's text may be a single word or
663            whitespace-separated words spoken by that speaker.
664        hyp: a list of (speaker, word) tuples of type (str, str) for the
665            hypothesis, same format as `ref`.
666        detailed: bool, if True, return a dict containing WDER, word counts
667            (wder_sub, wder_correct, wder_total), WER breakdown, and the
668            optimal speaker_mapping.
669
670    Returns:
671        If `detailed=False`, a float number for the Word Diarization Error
672        Rate.
673        If `detailed=True`, a dict with keys:
674            - "wder": float, Word Diarization Error Rate
675            - "wder_sub": int, aligned words with wrong speaker label
676            - "wder_correct": int, aligned words with correct speaker label
677            - "wder_total": int, total aligned words (wer_correct + wer_sub)
678            - "wer": float, Word Error Rate of the transcript
679            - "wer_insert": int, inserted words
680            - "wer_delete": int, deleted words
681            - "wer_sub": int, substituted words
682            - "wer_correct": int, correct words
683            - "wer_total": int, total reference words
684            - "speaker_mapping": dict mapping matched ref -> hyp speaker
685    """
686    if not isinstance(detailed, bool):
687        raise TypeError("detailed must be a boolean.")
688    check_word_input(ref)
689    check_word_input(hyp)
690
691    ref_expanded = _expand_word_tuples(ref)
692    hyp_expanded = _expand_word_tuples(hyp)
693
694    ref_spks = [spk for spk, _ in ref_expanded]
695    ref_words = [w for _, w in ref_expanded]
696    hyp_spks = [spk for spk, _ in hyp_expanded]
697    hyp_words = [w for _, w in hyp_expanded]
698
699    _, align = levenshtein_with_edits(ref_words, hyp_words)
700
701    wer_insert = 0
702    wer_delete = 0
703    wer_sub = 0
704    wer_correct = 0
705    aligned_spk_pairs = []
706
707    for i, j in align:
708        if i == -1:
709            wer_insert += 1
710        elif j == -1:
711            wer_delete += 1
712        else:
713            if ref_words[i] == hyp_words[j]:
714                wer_correct += 1
715            else:
716                wer_sub += 1
717            aligned_spk_pairs.append((ref_spks[i], hyp_spks[j]))
718
719    wer_total = len(ref_words)
720    wer_val = (
721        0.0 if wer_total == 0
722        else float((wer_insert + wer_delete + wer_sub) / wer_total)
723    )
724
725    wder_total = len(aligned_spk_pairs)
726    if wder_total == 0:
727        wder_correct = 0
728        wder_sub = 0
729        wder_val = 0.0
730        speaker_mapping = {}
731    else:
732        ref_index = build_speaker_index(ref_expanded)
733        hyp_index = build_speaker_index(hyp_expanded)
734        cost_matrix = np.zeros((len(ref_index), len(hyp_index)), dtype=int)
735        for r_spk, h_spk in aligned_spk_pairs:
736            cost_matrix[ref_index[r_spk], hyp_index[h_spk]] += 1
737
738        row_index, col_index = optimize.linear_sum_assignment(
739            cost_matrix, maximize=True)
740        wder_correct = int(cost_matrix[row_index, col_index].sum())
741        wder_sub = wder_total - wder_correct
742        wder_val = float(wder_sub / wder_total)
743
744        inv_ref = {i: spk for spk, i in ref_index.items()}
745        inv_hyp = {j: spk for spk, j in hyp_index.items()}
746        speaker_mapping = {
747            inv_ref[r]: inv_hyp[c]
748            for r, c in zip(row_index, col_index)
749            if cost_matrix[r, c] > 0
750        }
751
752    if detailed:
753        return {
754            "wder": wder_val,
755            "wder_sub": wder_sub,
756            "wder_correct": wder_correct,
757            "wder_total": wder_total,
758            "wer": wer_val,
759            "wer_insert": wer_insert,
760            "wer_delete": wer_delete,
761            "wer_sub": wer_sub,
762            "wer_correct": wer_correct,
763            "wer_total": wer_total,
764            "speaker_mapping": speaker_mapping,
765        }
766    return wder_val
