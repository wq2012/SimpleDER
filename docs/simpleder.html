363    """Compute Diarization Error Rate (DER).
364
365    Args:
366        ref: a list of tuples for the ground truth, where each tuple is
367            (speaker, start, end) of type (string, float, float)
368        hyp: a list of tuples for the diarization result hypothesis, same type
369            as `ref`
370        collar: float, no-score zone (in seconds) around each reference
371            speaker boundary ([boundary - collar, boundary + collar])
372        skip_overlap: bool, if True, exclude regions where more than one
373            reference speaker is active
374        detailed: bool, if True, return a dict containing DER, error component
375            durations (miss, false_alarm, confusion, ref_length), and the
376            optimal speaker_mapping
377
378    Returns:
379        If `detailed=False`, a float number for the Diarization Error Rate.
380        If `detailed=True`, a dict with keys:
381            - "der": float, Diarization Error Rate
382            - "miss": float, missed speech duration in seconds
383            - "false_alarm": float, false alarm duration in seconds
384            - "confusion": float, speaker confusion duration in seconds
385            - "ref_length": float, scored reference length in seconds
386            - "speaker_mapping": dict mapping matched ref -> hyp speaker
387    """
388    if not isinstance(detailed, bool):
389        raise TypeError("detailed must be a boolean.")
390    ref, hyp = _preprocess_segments(
391        ref, hyp, collar=collar, skip_overlap=skip_overlap)
392
393    ref_total_length = compute_total_length(ref)
394    ref_index = build_speaker_index(ref)
395    hyp_index = build_speaker_index(hyp)
396    cost_matrix = build_cost_matrix(ref, hyp)
397    row_index, col_index = optimize.linear_sum_assignment(-cost_matrix)
398    optimal_match_overlap = float(cost_matrix[row_index, col_index].sum())
399
400    inv_ref = {i: spk for spk, i in ref_index.items()}
401    inv_hyp = {j: spk for spk, j in hyp_index.items()}
402    speaker_mapping = {
403        inv_ref[r]: inv_hyp[c]
404        for r, c in zip(row_index, col_index)
405        if cost_matrix[r, c] > 0.0
406    }
407
408    miss, false_alarm, min_overlap, load_length = compute_error_components(
409        ref, hyp)
410    confusion = max(0.0, min_overlap - optimal_match_overlap)
411
412    if ref_total_length == 0.0:
413        der_val = 0.0
414    else:
415        der_val = (load_length - optimal_match_overlap) / ref_total_length
416
417    if detailed:
418        return {
419            "der": float(der_val),
420            "miss": float(miss),
421            "false_alarm": float(false_alarm),
422            "confusion": float(confusion),
423            "ref_length": float(ref_total_length),
424            "speaker_mapping": speaker_mapping,
425        }
426    return float(der_val)

429def JER(ref, hyp, collar=0.0, skip_overlap=False, detailed=False):
430    """Compute Jaccard Error Rate (JER).
431
432    JER is the average over all reference speakers of (1 - Jaccard similarity)
433    under an optimal 1-to-1 mapping between reference and hypothesis speakers:
434
435        JER_r = 1 - |r \\cap h| / |r \\cup h|
436        JER = (1 / |S_ref|) * \\sum_{r \\in S_ref} JER_r
437
438    Unmapped reference speakers have JER_r = 1.0.
439
440    Args:
441        ref: a list of tuples for the ground truth, where each tuple is
442            (speaker, start, end) of type (string, float, float)
443        hyp: a list of tuples for the diarization result hypothesis, same type
444            as `ref`
445        collar: float, no-score zone (in seconds) around each reference
446            speaker boundary ([boundary - collar, boundary + collar])
447        skip_overlap: bool, if True, exclude regions where more than one
448            reference speaker is active
449        detailed: bool, if True, return a dict containing overall JER,
450            per-reference-speaker JER, and the optimal speaker_mapping
451
452    Returns:
453        If `detailed=False`, a float number for the Jaccard Error Rate.
454        If `detailed=True`, a dict with keys:
455            - "jer": float, Jaccard Error Rate
456            - "speaker_jer": dict mapping each ref speaker -> speaker JER
457            - "speaker_mapping": dict mapping matched ref -> hyp speaker
458    """
459    if not isinstance(detailed, bool):
460        raise TypeError("detailed must be a boolean.")
461    ref, hyp = _preprocess_segments(
462        ref, hyp, collar=collar, skip_overlap=skip_overlap)
463
464    ref_index = build_speaker_index(ref)
465    hyp_index = build_speaker_index(hyp)
466
467    if not ref_index:
468        if detailed:
469            return {
470                "jer": 0.0,
471                "speaker_jer": {},
472                "speaker_mapping": {},
473            }
474        return 0.0
475
476    ref_durations = np.zeros(len(ref_index))
477    for spk, start, end in ref:
478        ref_durations[ref_index[spk]] += end - start
479
480    hyp_durations = np.zeros(len(hyp_index))
481    for spk, start, end in hyp:
482        hyp_durations[hyp_index[spk]] += end - start
483
484    intersection_matrix = build_cost_matrix(ref, hyp)
485    union_matrix = (
486        ref_durations[:, None] + hyp_durations[None, :] - intersection_matrix
487    )
488    jaccard_matrix = np.divide(
489        intersection_matrix,
490        union_matrix,
491        out=np.zeros_like(intersection_matrix),
492        where=union_matrix > 0.0,
493    )
494
495    row_index, col_index = optimize.linear_sum_assignment(-jaccard_matrix)
496
497    inv_ref = {i: spk for spk, i in ref_index.items()}
498    inv_hyp = {j: spk for spk, j in hyp_index.items()}
499    speaker_jer = {spk: 1.0 for spk in ref_index}
500    speaker_mapping = {}
501
502    for r, c in zip(row_index, col_index):
503        ref_spk = inv_ref[r]
504        speaker_jer[ref_spk] = float(1.0 - jaccard_matrix[r, c])
505        if intersection_matrix[r, c] > 0.0:
506            speaker_mapping[ref_spk] = inv_hyp[c]
507
508    jer_val = float(sum(speaker_jer.values()) / len(ref_index))
509
510    if detailed:
511        return {
512            "jer": jer_val,
513            "speaker_jer": speaker_jer,
514            "speaker_mapping": speaker_mapping,
515        }
516    return jer_val

Compute Jaccard Error Rate (JER).

JER is the average over all reference speakers of (1 - Jaccard similarity) under an optimal 1-to-1 mapping between reference and hypothesis speakers:

JER_r = 1 - |r \cap h| / |r \cup h|
JER = (1 / |S_ref|) * \sum_{r \in S_ref} JER_r

Unmapped reference speakers have JER_r = 1.0.

Args: ref: a list of tuples for the ground truth, where each tuple is (speaker, start, end) of type (string, float, float) hyp: a list of tuples for the diarization result hypothesis, same type as ref collar: float, no-score zone (in seconds) around each reference speaker boundary ([boundary - collar, boundary + collar]) skip_overlap: bool, if True, exclude regions where more than one reference speaker is active detailed: bool, if True, return a dict containing overall JER, per-reference-speaker JER, and the optimal speaker_mapping

Returns: If detailed=False, a float number for the Jaccard Error Rate. If detailed=True, a dict with keys: - "jer": float, Jaccard Error Rate - "speaker_jer": dict mapping each ref speaker -> speaker JER - "speaker_mapping": dict mapping matched ref -> hyp speaker

536def corpus_der(refs, hyps, collar=0.0, skip_overlap=False, detailed=False):
537    """Compute micro-averaged Corpus Diarization Error Rate across recordings.
538
539    Corpus DER aggregates total missed speech, false alarm, and speaker
540    confusion durations across all recordings and divides by the total
541    reference duration across all recordings.
542
543    Args:
544        refs: a dict mapping recording_id -> ref segments, or a list of
545            ref segment lists
546        hyps: a dict mapping recording_id -> hyp segments, or a list of
547            hyp segment lists (matching `refs`)
548        collar: float, no-score zone (in seconds) around each reference
549            speaker boundary
550        skip_overlap: bool, if True, exclude regions where more than one
551            reference speaker is active
552        detailed: bool, if True, return a dict with corpus-level totals and
553            per-recording results
554
555    Returns:
556        If `detailed=False`, a float number for the Corpus DER.
557        If `detailed=True`, a dict with keys:
558            - "der": float, Corpus DER
559            - "miss": float, total missed speech duration in seconds
560            - "false_alarm": float, total false alarm duration in seconds
561            - "confusion": float, total speaker confusion duration in seconds
562            - "ref_length": float, total scored reference length in seconds
563            - "per_recording": dict or list of per-recording detailed results
564    """
565    if not isinstance(detailed, bool):
566        raise TypeError("detailed must be a boolean.")
567    pairs, is_dict = _normalize_corpus_inputs(refs, hyps)
568
569    total_miss = 0.0
570    total_fa = 0.0
571    total_confusion = 0.0
572    total_ref_length = 0.0
573    per_recording = {}
574
575    for key, ref, hyp in pairs:
576        res = DER(
577            ref, hyp, collar=collar, skip_overlap=skip_overlap, detailed=True)
578        total_miss += res["miss"]
579        total_fa += res["false_alarm"]
580        total_confusion += res["confusion"]
581        total_ref_length += res["ref_length"]
582        per_recording[key] = res
583
584    if total_ref_length == 0.0:
585        der_val = 0.0
586    else:
587        der_val = (total_miss + total_fa + total_confusion) / total_ref_length
588
589    if detailed:
590        return {
591            "der": float(der_val),
592            "miss": float(total_miss),
593            "false_alarm": float(total_fa),
594            "confusion": float(total_confusion),
595            "ref_length": float(total_ref_length),
596            "per_recording": per_recording if is_dict else [
597                per_recording[i] for i in range(len(pairs))
598            ],
599        }
600    return float(der_val)

603def corpus_jer(refs, hyps, collar=0.0, skip_overlap=False, detailed=False):
604    """Compute Corpus Jaccard Error Rate across multiple recordings.
605
606    Corpus JER is the average of per-speaker Jaccard Error Rates across all
607    reference speakers in all recordings.
608
609    Args:
610        refs: a dict mapping recording_id -> ref segments, or a list of
611            ref segment lists
612        hyps: a dict mapping recording_id -> hyp segments, or a list of
613            hyp segment lists (matching `refs`)
614        collar: float, no-score zone (in seconds) around each reference
615            speaker boundary
616        skip_overlap: bool, if True, exclude regions where more than one
617            reference speaker is active
618        detailed: bool, if True, return a dict with corpus JER, total number of
619            reference speakers, and per-recording results
620
621    Returns:
622        If `detailed=False`, a float number for the Corpus JER.
623        If `detailed=True`, a dict with keys:
624            - "jer": float, Corpus JER
625            - "num_ref_speakers": int, total reference speakers across corpus
626            - "per_recording": dict or list of per-recording detailed results
627    """
628    if not isinstance(detailed, bool):
629        raise TypeError("detailed must be a boolean.")
630    pairs, is_dict = _normalize_corpus_inputs(refs, hyps)
631
632    total_speaker_jer = 0.0
633    total_ref_speakers = 0
634    per_recording = {}
635
636    for key, ref, hyp in pairs:
637        res = JER(
638            ref, hyp, collar=collar, skip_overlap=skip_overlap, detailed=True)
639        total_speaker_jer += sum(res["speaker_jer"].values())
640        total_ref_speakers += len(res["speaker_jer"])
641        per_recording[key] = res
642
643    if total_ref_speakers == 0:
644        jer_val = 0.0
645    else:
646        jer_val = total_speaker_jer / total_ref_speakers
647
648    if detailed:
649        return {
650            "jer": float(jer_val),
651            "num_ref_speakers": total_ref_speakers,
652            "per_recording": per_recording if is_dict else [
653                per_recording[i] for i in range(len(pairs))
654            ],
655        }
656    return float(jer_val)
