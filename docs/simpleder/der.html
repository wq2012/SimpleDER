  5def check_input(hyp):
  6    """Check whether a hypothesis/reference is valid.
  7
  8    Args:
  9        hyp: a list of tuples, where each tuple is (speaker, start, end)
 10            of type (string, float, float)
 11
 12    Raises:
 13        TypeError: if the type of `hyp` or its elements is incorrect
 14        ValueError: if some tuple has non-finite timestamps or start > end
 15    """
 16    if not isinstance(hyp, list):
 17        raise TypeError("Input must be a list.")
 18    for element in hyp:
 19        if not isinstance(element, tuple):
 20            raise TypeError("Input must be a list of tuples.")
 21        if len(element) != 3:
 22            raise TypeError(
 23                "Each tuple must have the elements: (speaker, start, end).")
 24        if not isinstance(element[0], str):
 25            raise TypeError("Speaker must be a string.")
 26        if (isinstance(element[1], bool) or isinstance(element[2], bool) or
 27                not isinstance(element[1], (int, float, np.number)) or
 28                not isinstance(element[2], (int, float, np.number))):
 29            raise TypeError("Start and end must be float numbers.")
 30        if not np.isfinite(element[1]) or not np.isfinite(element[2]):
 31            raise ValueError("Start and end must be finite numbers.")
 32        if element[1] > element[2]:
 33            raise ValueError("Start must not be larger than end.")

 36def _merge_intervals(intervals):
 37    """Sort and merge a list of (start, end) intervals."""
 38    if not intervals:
 39        return []
 40    sorted_intervals = sorted(intervals)
 41    merged = []
 42    curr_start, curr_end = sorted_intervals[0]
 43    for next_start, next_end in sorted_intervals[1:]:
 44        if next_start <= curr_end:
 45            curr_end = max(curr_end, next_end)
 46        else:
 47            merged.append((curr_start, curr_end))
 48            curr_start, curr_end = next_start, next_end
 49    merged.append((curr_start, curr_end))
 50    return merged

 53def merge_speaker_intervals(segments):
 54    """Merge overlapping or contiguous intervals for each speaker.
 55
 56    Also removes zero-length intervals where start == end.
 57
 58    Args:
 59        segments: a list of (speaker, start, end) tuples
 60
 61    Returns:
 62        a list of (speaker, start, end) tuples with disjoint, non-adjacent
 63        intervals per speaker, sorted by (start, end, speaker)
 64    """
 65    by_speaker = {}
 66    for speaker, start, end in segments:
 67        start_f = float(start)
 68        end_f = float(end)
 69        if end_f <= start_f:
 70            continue
 71        by_speaker.setdefault(speaker, []).append((start_f, end_f))
 72
 73    merged_segments = []
 74    for speaker, intervals in by_speaker.items():
 75        for start_f, end_f in _merge_intervals(intervals):
 76            merged_segments.append((speaker, start_f, end_f))
 77
 78    merged_segments.sort(key=lambda x: (x[1], x[2], x[0]))
 79    return merged_segments

 82def compute_total_length(hyp):
 83    """Compute total length of a hypothesis/reference.
 84
 85    Overlapping intervals belonging to the same speaker are merged before
 86    summing durations.
 87
 88    Args:
 89        hyp: a list of tuples, where each tuple is (speaker, start, end)
 90            of type (string, float, float)
 91
 92    Returns:
 93        a float number for the total length
 94    """
 95    merged_hyp = merge_speaker_intervals(hyp)
 96    total_length = 0.0
 97    for element in merged_hyp:
 98        total_length += element[2] - element[1]
 99    return total_length

102def compute_intersection_length(A, B):
103    """Compute the intersection length of two tuples.
104
105    Args:
106        A: a (speaker, start, end) tuple of type (string, float, float)
107        B: a (speaker, start, end) tuple of type (string, float, float)
108
109    Returns:
110        a float number of the intersection between `A` and `B`
111    """
112    max_start = max(A[1], B[1])
113    min_end = min(A[2], B[2])
114    return max(0.0, min_end - max_start)

117def compute_error_components(ref, hyp):
118    """Compute elementary duration components between reference and hypothesis.
119
120    Args:
121        ref: a list of (speaker, start, end) tuples for the ground truth
122        hyp: a list of (speaker, start, end) tuples for the hypothesis
123
124    Returns:
125        a tuple of (miss, false_alarm, min_overlap, load_length) in seconds,
126        where:
127            miss = \\int max(0, N_ref(t) - N_hyp(t)) dt
128            false_alarm = \\int max(0, N_hyp(t) - N_ref(t)) dt
129            min_overlap = \\int min(N_ref(t), N_hyp(t)) dt
130            load_length = \\int max(N_ref(t), N_hyp(t)) dt
131    """
132    ref = merge_speaker_intervals(ref)
133    hyp = merge_speaker_intervals(hyp)
134
135    boundaries = set()
136    for element in ref + hyp:
137        boundaries.add(element[1])
138        boundaries.add(element[2])
139    boundaries = sorted(list(boundaries))
140
141    miss = 0.0
142    false_alarm = 0.0
143    min_overlap = 0.0
144    load_length = 0.0
145
146    for i in range(len(boundaries) - 1):
147        start = boundaries[i]
148        end = boundaries[i + 1]
149        dur = end - start
150
151        ref_count = 0
152        for element in ref:
153            if element[1] <= start and end <= element[2]:
154                ref_count += 1
155
156        hyp_count = 0
157        for element in hyp:
158            if element[1] <= start and end <= element[2]:
159                hyp_count += 1
160
161        miss += dur * max(0, ref_count - hyp_count)
162        false_alarm += dur * max(0, hyp_count - ref_count)
163        min_overlap += dur * min(ref_count, hyp_count)
164        load_length += dur * max(ref_count, hyp_count)
165
166    return miss, false_alarm, min_overlap, load_length

169def compute_load_length(ref, hyp):
170    """Compute the load length (integrated maximum number of speakers).
171
172    The "load" generalizes the union of reference and hypothesis when
173    overlapped speech is present. It is defined as the integral of the
174    maximum number of speakers at any given time point:
175
176        Load = \\int max(N_ref(t), N_hyp(t)) dt
177
178    where N_ref(t) is the number of speakers in reference at time t, and
179    N_hyp(t) is the number of speakers in hypothesis at time t.
180
181    Args:
182        ref: a list of tuples for the ground truth, where each tuple is
183            (speaker, start, end) of type (string, float, float)
184        hyp: a list of tuples for the diarization result hypothesis, same type
185            as `ref`
186
187    Returns:
188        a float number for the load length
189    """
190    _, _, _, load_length = compute_error_components(ref, hyp)
191    return load_length

209def build_cost_matrix(ref, hyp):
210    """Build the cost (intersection) matrix between reference and hypothesis.
211
212    Args:
213        ref: a list of tuples for the ground truth, where each tuple is
214            (speaker, start, end) of type (string, float, float)
215        hyp: a list of tuples for the diarization result hypothesis, same type
216            as `ref`
217
218    Returns:
219        a 2-dim numpy array, whose element (i, j) is the overlap duration
220        between `i`th reference speaker and `j`th hypothesis speaker
221    """
222    ref = merge_speaker_intervals(ref)
223    hyp = merge_speaker_intervals(hyp)
224    ref_index = build_speaker_index(ref)
225    hyp_index = build_speaker_index(hyp)
226    cost_matrix = np.zeros((len(ref_index), len(hyp_index)))
227    for ref_element in ref:
228        for hyp_element in hyp:
229            i = ref_index[ref_element[0]]
230            j = hyp_index[hyp_element[0]]
231            cost_matrix[i, j] += compute_intersection_length(
232                ref_element, hyp_element)
233    return cost_matrix

236def compute_merged_exclusion_intervals(ref, collar):
237    """Compute merged exclusion intervals based on reference boundaries.
238
239    Args:
240        ref: a list of tuples for the ground truth
241        collar: float, no-score tolerance around each reference boundary
242            ([boundary - collar, boundary + collar])
243
244    Returns:
245        a list of (start, end) tuples, sorted and merged
246    """
247    if isinstance(collar, bool) or not isinstance(
248            collar, (int, float, np.number)):
249        raise TypeError("Collar must be a number.")
250    if not np.isfinite(collar) or collar < 0.0:
251        raise ValueError("Collar must be a non-negative finite number.")
252    collar = float(collar)
253    if collar == 0.0:
254        return []
255
256    ref = merge_speaker_intervals(ref)
257    intervals = []
258    for element in ref:
259        start, end = element[1], element[2]
260        intervals.append((start - collar, start + collar))
261        intervals.append((end - collar, end + collar))
262
263    return _merge_intervals(intervals)

266def compute_overlap_exclusion_intervals(ref):
267    """Compute merged intervals where multiple reference speakers are active.
268
269    Args:
270        ref: a list of (speaker, start, end) tuples for the ground truth
271
272    Returns:
273        a list of (start, end) tuples where N_ref(t) > 1, sorted and merged
274    """
275    ref = merge_speaker_intervals(ref)
276    boundaries = set()
277    for element in ref:
278        boundaries.add(element[1])
279        boundaries.add(element[2])
280    boundaries = sorted(list(boundaries))
281
282    overlap_intervals = []
283    for i in range(len(boundaries) - 1):
284        start = boundaries[i]
285        end = boundaries[i + 1]
286        ref_count = 0
287        for element in ref:
288            if element[1] <= start and end <= element[2]:
289                ref_count += 1
290        if ref_count > 1:
291            overlap_intervals.append((start, end))
292
293    return _merge_intervals(overlap_intervals)

296def subtract_intervals(segments, exclusions):
297    """Subtract exclusion intervals from segments.
298
299    Args:
300        segments: a list of (speaker, start, end)
301        exclusions: a list of (start, end) tuples, sorted and merged
302
303    Returns:
304        a list of (speaker, start, end) with exclusions removed
305    """
306    if not exclusions:
307        return segments
308
309    new_segments = []
310    for speaker, start, end in segments:
311        current_time = start
312        for ex_start, ex_end in exclusions:
313            if ex_end <= current_time:
314                continue
315            if ex_start >= end:
316                break
317
318            if ex_start > current_time:
319                new_segments.append((speaker, current_time, ex_start))
320
321            current_time = max(current_time, ex_end)
322
323            if current_time >= end:
324                break
325
326        if current_time < end:
327            new_segments.append((speaker, current_time, end))
328
329    return new_segments

332def _preprocess_segments(ref, hyp, collar=0.0, skip_overlap=False):
333    """Validate and preprocess reference and hypothesis segments."""
334    check_input(ref)
335    check_input(hyp)
336    if not isinstance(skip_overlap, bool):
337        raise TypeError("skip_overlap must be a boolean.")
338    if not isinstance(collar, bool) and isinstance(
339            collar, (int, float, np.number)):
340        if not np.isfinite(collar) or collar < 0.0:
341            raise ValueError("Collar must be a non-negative finite number.")
342    else:
343        raise TypeError("Collar must be a number.")
344
345    ref = merge_speaker_intervals(ref)
346    hyp = merge_speaker_intervals(hyp)
347
348    exclusions = []
349    if collar > 0.0:
350        exclusions.extend(compute_merged_exclusion_intervals(ref, collar))
351    if skip_overlap:
352        exclusions.extend(compute_overlap_exclusion_intervals(ref))
353    if exclusions:
354        exclusions = _merge_intervals(exclusions)
355        ref = subtract_intervals(ref, exclusions)
356        hyp = subtract_intervals(hyp, exclusions)
357
358    return ref, hyp

361def DER(ref, hyp, collar=0.0, skip_overlap=False, detailed=False):
362    """Compute Diarization Error Rate (DER).
363
364    Args:
365        ref: a list of tuples for the ground truth, where each tuple is
366            (speaker, start, end) of type (string, float, float)
367        hyp: a list of tuples for the diarization result hypothesis, same type
368            as `ref`
369        collar: float, no-score zone (in seconds) around each reference
370            speaker boundary ([boundary - collar, boundary + collar])
371        skip_overlap: bool, if True, exclude regions where more than one
372            reference speaker is active
373        detailed: bool, if True, return a dict containing DER, error component
374            durations (miss, false_alarm, confusion, ref_length), and the
375            optimal speaker_mapping
376
377    Returns:
378        If `detailed=False`, a float number for the Diarization Error Rate.
379        If `detailed=True`, a dict with keys:
380            - "der": float, Diarization Error Rate
381            - "miss": float, missed speech duration in seconds
382            - "false_alarm": float, false alarm duration in seconds
383            - "confusion": float, speaker confusion duration in seconds
384            - "ref_length": float, scored reference length in seconds
385            - "speaker_mapping": dict mapping matched ref -> hyp speaker
386    """
387    if not isinstance(detailed, bool):
388        raise TypeError("detailed must be a boolean.")
389    ref, hyp = _preprocess_segments(
390        ref, hyp, collar=collar, skip_overlap=skip_overlap)
391
392    ref_total_length = compute_total_length(ref)
393    ref_index = build_speaker_index(ref)
394    hyp_index = build_speaker_index(hyp)
395    cost_matrix = build_cost_matrix(ref, hyp)
396    row_index, col_index = optimize.linear_sum_assignment(-cost_matrix)
397    optimal_match_overlap = float(cost_matrix[row_index, col_index].sum())
398
399    inv_ref = {i: spk for spk, i in ref_index.items()}
400    inv_hyp = {j: spk for spk, j in hyp_index.items()}
401    speaker_mapping = {
402        inv_ref[r]: inv_hyp[c]
403        for r, c in zip(row_index, col_index)
404        if cost_matrix[r, c] > 0.0
405    }
406
407    miss, false_alarm, min_overlap, load_length = compute_error_components(
408        ref, hyp)
409    confusion = max(0.0, min_overlap - optimal_match_overlap)
410
411    if ref_total_length == 0.0:
412        der_val = 0.0
413    else:
414        der_val = (load_length - optimal_match_overlap) / ref_total_length
415
416    if detailed:
417        return {
418            "der": float(der_val),
419            "miss": float(miss),
420            "false_alarm": float(false_alarm),
421            "confusion": float(confusion),
422            "ref_length": float(ref_total_length),
423            "speaker_mapping": speaker_mapping,
424        }
425    return float(der_val)

428def JER(ref, hyp, collar=0.0, skip_overlap=False, detailed=False):
429    """Compute Jaccard Error Rate (JER).
430
431    JER is the average over all reference speakers of (1 - Jaccard similarity)
432    under an optimal 1-to-1 mapping between reference and hypothesis speakers:
433
434        JER_r = 1 - |r \\cap h| / |r \\cup h|
435        JER = (1 / |S_ref|) * \\sum_{r \\in S_ref} JER_r
436
437    Unmapped reference speakers have JER_r = 1.0.
438
439    Args:
440        ref: a list of tuples for the ground truth, where each tuple is
441            (speaker, start, end) of type (string, float, float)
442        hyp: a list of tuples for the diarization result hypothesis, same type
443            as `ref`
444        collar: float, no-score zone (in seconds) around each reference
445            speaker boundary ([boundary - collar, boundary + collar])
446        skip_overlap: bool, if True, exclude regions where more than one
447            reference speaker is active
448        detailed: bool, if True, return a dict containing overall JER,
449            per-reference-speaker JER, and the optimal speaker_mapping
450
451    Returns:
452        If `detailed=False`, a float number for the Jaccard Error Rate.
453        If `detailed=True`, a dict with keys:
454            - "jer": float, Jaccard Error Rate
455            - "speaker_jer": dict mapping each ref speaker -> speaker JER
456            - "speaker_mapping": dict mapping matched ref -> hyp speaker
457    """
458    if not isinstance(detailed, bool):
459        raise TypeError("detailed must be a boolean.")
460    ref, hyp = _preprocess_segments(
461        ref, hyp, collar=collar, skip_overlap=skip_overlap)
462
463    ref_index = build_speaker_index(ref)
464    hyp_index = build_speaker_index(hyp)
465
466    if not ref_index:
467        if detailed:
468            return {
469                "jer": 0.0,
470                "speaker_jer": {},
471                "speaker_mapping": {},
472            }
473        return 0.0
474
475    ref_durations = np.zeros(len(ref_index))
476    for spk, start, end in ref:
477        ref_durations[ref_index[spk]] += end - start
478
479    hyp_durations = np.zeros(len(hyp_index))
480    for spk, start, end in hyp:
481        hyp_durations[hyp_index[spk]] += end - start
482
483    intersection_matrix = build_cost_matrix(ref, hyp)
484    union_matrix = (
485        ref_durations[:, None] + hyp_durations[None, :] - intersection_matrix
486    )
487    jaccard_matrix = np.divide(
488        intersection_matrix,
489        union_matrix,
490        out=np.zeros_like(intersection_matrix),
491        where=union_matrix > 0.0,
492    )
493
494    row_index, col_index = optimize.linear_sum_assignment(-jaccard_matrix)
495
496    inv_ref = {i: spk for spk, i in ref_index.items()}
497    inv_hyp = {j: spk for spk, j in hyp_index.items()}
498    speaker_jer = {spk: 1.0 for spk in ref_index}
499    speaker_mapping = {}
500
501    for r, c in zip(row_index, col_index):
502        ref_spk = inv_ref[r]
503        speaker_jer[ref_spk] = float(1.0 - jaccard_matrix[r, c])
504        if intersection_matrix[r, c] > 0.0:
505            speaker_mapping[ref_spk] = inv_hyp[c]
506
507    jer_val = float(sum(speaker_jer.values()) / len(ref_index))
508
509    if detailed:
510        return {
511            "jer": jer_val,
512            "speaker_jer": speaker_jer,
513            "speaker_mapping": speaker_mapping,
514        }
515    return jer_val

518def _normalize_corpus_inputs(refs, hyps):
519    """Validate and normalize multi-recording inputs (dicts or lists)."""
520    if isinstance(refs, dict) and isinstance(hyps, dict):
521        if set(refs.keys()) != set(hyps.keys()):
522            raise ValueError(
523                "Reference and hypothesis dicts must have the same keys.")
524        pairs = [(k, refs[k], hyps[k]) for k in sorted(refs.keys())]
525        return pairs, True
526    if isinstance(refs, list) and isinstance(hyps, list):
527        if len(refs) != len(hyps):
528            raise ValueError(
529                "Reference and hypothesis lists must have the same length.")
530        pairs = [(i, refs[i], hyps[i]) for i in range(len(refs))]
531        return pairs, False
532    raise TypeError("refs and hyps must both be dicts or both be lists.")

535def corpus_der(refs, hyps, collar=0.0, skip_overlap=False, detailed=False):
536    """Compute micro-averaged Corpus Diarization Error Rate across recordings.
537
538    Corpus DER aggregates total missed speech, false alarm, and speaker
539    confusion durations across all recordings and divides by the total
540    reference duration across all recordings.
541
542    Args:
543        refs: a dict mapping recording_id -> ref segments, or a list of
544            ref segment lists
545        hyps: a dict mapping recording_id -> hyp segments, or a list of
546            hyp segment lists (matching `refs`)
547        collar: float, no-score zone (in seconds) around each reference
548            speaker boundary
549        skip_overlap: bool, if True, exclude regions where more than one
550            reference speaker is active
551        detailed: bool, if True, return a dict with corpus-level totals and
552            per-recording results
553
554    Returns:
555        If `detailed=False`, a float number for the Corpus DER.
556        If `detailed=True`, a dict with keys:
557            - "der": float, Corpus DER
558            - "miss": float, total missed speech duration in seconds
559            - "false_alarm": float, total false alarm duration in seconds
560            - "confusion": float, total speaker confusion duration in seconds
561            - "ref_length": float, total scored reference length in seconds
562            - "per_recording": dict or list of per-recording detailed results
563    """
564    if not isinstance(detailed, bool):
565        raise TypeError("detailed must be a boolean.")
566    pairs, is_dict = _normalize_corpus_inputs(refs, hyps)
567
568    total_miss = 0.0
569    total_fa = 0.0
570    total_confusion = 0.0
571    total_ref_length = 0.0
572    per_recording = {}
573
574    for key, ref, hyp in pairs:
575        res = DER(
576            ref, hyp, collar=collar, skip_overlap=skip_overlap, detailed=True)
577        total_miss += res["miss"]
578        total_fa += res["false_alarm"]
579        total_confusion += res["confusion"]
580        total_ref_length += res["ref_length"]
581        per_recording[key] = res
582
583    if total_ref_length == 0.0:
584        der_val = 0.0
585    else:
586        der_val = (total_miss + total_fa + total_confusion) / total_ref_length
587
588    if detailed:
589        return {
590            "der": float(der_val),
591            "miss": float(total_miss),
592            "false_alarm": float(total_fa),
593            "confusion": float(total_confusion),
594            "ref_length": float(total_ref_length),
595            "per_recording": per_recording if is_dict else [
596                per_recording[i] for i in range(len(pairs))
597            ],
598        }
599    return float(der_val)

602def corpus_jer(refs, hyps, collar=0.0, skip_overlap=False, detailed=False):
603    """Compute Corpus Jaccard Error Rate across multiple recordings.
604
605    Corpus JER is the average of per-speaker Jaccard Error Rates across all
606    reference speakers in all recordings.
607
608    Args:
609        refs: a dict mapping recording_id -> ref segments, or a list of
610            ref segment lists
611        hyps: a dict mapping recording_id -> hyp segments, or a list of
612            hyp segment lists (matching `refs`)
613        collar: float, no-score zone (in seconds) around each reference
614            speaker boundary
615        skip_overlap: bool, if True, exclude regions where more than one
616            reference speaker is active
617        detailed: bool, if True, return a dict with corpus JER, total number of
618            reference speakers, and per-recording results
619
620    Returns:
621        If `detailed=False`, a float number for the Corpus JER.
622        If `detailed=True`, a dict with keys:
623            - "jer": float, Corpus JER
624            - "num_ref_speakers": int, total reference speakers across corpus
625            - "per_recording": dict or list of per-recording detailed results
626    """
627    if not isinstance(detailed, bool):
628        raise TypeError("detailed must be a boolean.")
629    pairs, is_dict = _normalize_corpus_inputs(refs, hyps)
630
631    total_speaker_jer = 0.0
632    total_ref_speakers = 0
633    per_recording = {}
634
635    for key, ref, hyp in pairs:
636        res = JER(
637            ref, hyp, collar=collar, skip_overlap=skip_overlap, detailed=True)

429def JER(ref, hyp, collar=0.0, skip_overlap=False, detailed=False):
430    """Compute Jaccard Error Rate (JER).
431
432    JER is the average over all reference speakers of (1 - Jaccard similarity)
433    under an optimal 1-to-1 mapping between reference and hypothesis speakers:
434
435        JER_r = 1 - |r \\cap h| / |r \\cup h|
436        JER = (1 / |S_ref|) * \\sum_{r \\in S_ref} JER_r
437
438    Unmapped reference speakers have JER_r = 1.0.
439
440    Args:
441        ref: a list of tuples for the ground truth, where each tuple is
442            (speaker, start, end) of type (string, float, float)
443        hyp: a list of tuples for the diarization result hypothesis, same type
444            as `ref`
445        collar: float, no-score zone (in seconds) around each reference
446            speaker boundary ([boundary - collar, boundary + collar])
447        skip_overlap: bool, if True, exclude regions where more than one
448            reference speaker is active
449        detailed: bool, if True, return a dict containing overall JER,
450            per-reference-speaker JER, and the optimal speaker_mapping
451
452    Returns:
453        If `detailed=False`, a float number for the Jaccard Error Rate.
454        If `detailed=True`, a dict with keys:
455            - "jer": float, Jaccard Error Rate
456            - "speaker_jer": dict mapping each ref speaker -> speaker JER
457            - "speaker_mapping": dict mapping matched ref -> hyp speaker
458    """
459    if not isinstance(detailed, bool):
460        raise TypeError("detailed must be a boolean.")
461    ref, hyp = _preprocess_segments(
462        ref, hyp, collar=collar, skip_overlap=skip_overlap)
463
464    ref_index = build_speaker_index(ref)
465    hyp_index = build_speaker_index(hyp)
466
467    if not ref_index:
468        if detailed:
469            return {
470                "jer": 0.0,
471                "speaker_jer": {},
472                "speaker_mapping": {},
473            }
474        return 0.0
475
476    ref_durations = np.zeros(len(ref_index))
477    for spk, start, end in ref:
478        ref_durations[ref_index[spk]] += end - start
479
480    hyp_durations = np.zeros(len(hyp_index))
481    for spk, start, end in hyp:
482        hyp_durations[hyp_index[spk]] += end - start
483
484    intersection_matrix = build_cost_matrix(ref, hyp)
485    union_matrix = (
486        ref_durations[:, None] + hyp_durations[None, :] - intersection_matrix
487    )
488    jaccard_matrix = np.divide(
489        intersection_matrix,
490        union_matrix,
491        out=np.zeros_like(intersection_matrix),
492        where=union_matrix > 0.0,
493    )
494
495    row_index, col_index = optimize.linear_sum_assignment(-jaccard_matrix)
496
497    inv_ref = {i: spk for spk, i in ref_index.items()}
498    inv_hyp = {j: spk for spk, j in hyp_index.items()}
499    speaker_jer = {spk: 1.0 for spk in ref_index}
500    speaker_mapping = {}
501
502    for r, c in zip(row_index, col_index):
503        ref_spk = inv_ref[r]
504        speaker_jer[ref_spk] = float(1.0 - jaccard_matrix[r, c])
505        if intersection_matrix[r, c] > 0.0:
506            speaker_mapping[ref_spk] = inv_hyp[c]
507
508    jer_val = float(sum(speaker_jer.values()) / len(ref_index))
509
510    if detailed:
511        return {
512            "jer": jer_val,
513            "speaker_jer": speaker_jer,
514            "speaker_mapping": speaker_mapping,
515        }
516    return jer_val
