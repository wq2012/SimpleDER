  16def check_input(hyp):
  17    """Check whether a hypothesis/reference segment list is valid.
  18
  19    Args:
  20        hyp: a list of tuples, where each tuple is (speaker, start, end)
  21            of type (string, float, float)
  22
  23    Raises:
  24        TypeError: if the type of `hyp` or its elements is incorrect
  25        ValueError: if some tuple has non-finite timestamps or start > end
  26    """
  27    if not isinstance(hyp, list):
  28        raise TypeError("Input must be a list.")
  29    for element in hyp:
  30        if not isinstance(element, tuple):
  31            raise TypeError("Input must be a list of tuples.")
  32        if len(element) != 3:
  33            raise TypeError(
  34                "Each tuple must have the elements: (speaker, start, end).")
  35        if not isinstance(element[0], str):
  36            raise TypeError("Speaker must be a string.")
  37        if (isinstance(element[1], bool) or isinstance(element[2], bool) or
  38                not isinstance(element[1], (int, float, np.number)) or
  39                not isinstance(element[2], (int, float, np.number))):
  40            raise TypeError("Start and end must be float numbers.")
  41        if not np.isfinite(element[1]) or not np.isfinite(element[2]):
  42            raise ValueError("Start and end must be finite numbers.")
  43        if element[1] > element[2]:
  44            raise ValueError("Start must not be larger than end.")

  47def _merge_intervals(intervals):
  48    """Sort and merge a list of (start, end) intervals."""
  49    if not intervals:
  50        return []
  51    sorted_intervals = sorted(intervals)
  52    merged = []
  53    curr_start, curr_end = sorted_intervals[0]
  54    for next_start, next_end in sorted_intervals[1:]:
  55        if next_start <= curr_end:
  56            curr_end = max(curr_end, next_end)
  57        else:
  58            merged.append((curr_start, curr_end))
  59            curr_start, curr_end = next_start, next_end
  60    merged.append((curr_start, curr_end))
  61    return merged

  64def merge_speaker_intervals(segments):
  65    """Merge overlapping or contiguous intervals for each speaker.
  66
  67    Also removes zero-length intervals where start == end.
  68
  69    Args:
  70        segments: a list of (speaker, start, end) tuples
  71
  72    Returns:
  73        a list of (speaker, start, end) tuples with disjoint, non-adjacent
  74        intervals per speaker, sorted by (start, end, speaker)
  75    """
  76    by_speaker = {}
  77    for speaker, start, end in segments:
  78        start_f = float(start)
  79        end_f = float(end)
  80        if end_f <= start_f:
  81            continue
  82        by_speaker.setdefault(speaker, []).append((start_f, end_f))
  83
  84    merged_segments = []
  85    for speaker, intervals in by_speaker.items():
  86        for start_f, end_f in _merge_intervals(intervals):
  87            merged_segments.append((speaker, start_f, end_f))
  88
  89    merged_segments.sort(key=lambda x: (x[1], x[2], x[0]))
  90    return merged_segments

  93def compute_total_length(hyp):
  94    """Compute total length of a hypothesis/reference.
  95
  96    Overlapping intervals belonging to the same speaker are merged before
  97    summing durations.
  98
  99    Args:
 100        hyp: a list of tuples, where each tuple is (speaker, start, end)
 101            of type (string, float, float)
 102
 103    Returns:
 104        a float number for the total length
 105    """
 106    merged_hyp = merge_speaker_intervals(hyp)
 107    total_length = 0.0
 108    for element in merged_hyp:
 109        total_length += element[2] - element[1]
 110    return total_length

 113def compute_intersection_length(A, B):
 114    """Compute the intersection length of two tuples.
 115
 116    Args:

 120    Returns:

 125    return max(0.0, min_end - max_start)

 128def compute_error_components(ref, hyp):
 129    """Compute elementary duration components between reference and hypothesis.
 130
 131    Args:
 132        ref: a list of (speaker, start, end) tuples for the ground truth
 133        hyp: a list of (speaker, start, end) tuples for the hypothesis
 134
 135    Returns:
 136        a tuple of (miss, false_alarm, min_overlap, load_length) in seconds,
 137        where:
 138            miss = \\int max(0, N_ref(t) - N_hyp(t)) dt
 139            false_alarm = \\int max(0, N_hyp(t) - N_ref(t)) dt
 140            min_overlap = \\int min(N_ref(t), N_hyp(t)) dt
 141            load_length = \\int max(N_ref(t), N_hyp(t)) dt
 142    """
 143    ref = merge_speaker_intervals(ref)
 144    hyp = merge_speaker_intervals(hyp)
 145
 146    boundaries = set()
 147    for element in ref + hyp:
 148        boundaries.add(element[1])
 149        boundaries.add(element[2])
 150    boundaries = sorted(list(boundaries))
 151
 152    miss = 0.0
 153    false_alarm = 0.0
 154    min_overlap = 0.0
 155    load_length = 0.0
 156
 157    for i in range(len(boundaries) - 1):
 158        start = boundaries[i]
 159        end = boundaries[i + 1]
 160        dur = end - start
 161
 162        ref_count = 0
 163        for element in ref:
 164            if element[1] <= start and end <= element[2]:
 165                ref_count += 1
 166
 167        hyp_count = 0
 168        for element in hyp:
 169            if element[1] <= start and end <= element[2]:
 170                hyp_count += 1
 171
 172        miss += dur * max(0, ref_count - hyp_count)
 173        false_alarm += dur * max(0, hyp_count - ref_count)
 174        min_overlap += dur * min(ref_count, hyp_count)
 175        load_length += dur * max(ref_count, hyp_count)
 176
 177    return miss, false_alarm, min_overlap, load_length

 180def compute_load_length(ref, hyp):
 181    """Compute the load length (integrated maximum number of speakers).
 182
 183    The "load" generalizes the union of reference and hypothesis when
 184    overlapped speech is present. It is defined as the integral of the
 185    maximum number of speakers at any given time point:
 186
 187        Load = \\int max(N_ref(t), N_hyp(t)) dt
 188
 189    where N_ref(t) is the number of speakers in reference at time t, and
 190    N_hyp(t) is the number of speakers in hypothesis at time t.
 191
 192    Args:
 193        ref: a list of tuples for the ground truth, where each tuple is
 194            (speaker, start, end) of type (string, float, float)
 195        hyp: a list of tuples for the diarization result hypothesis, same type
 196            as `ref`
 197
 198    Returns:
 199        a float number for the load length
 200    """
 201    _, _, _, load_length = compute_error_components(ref, hyp)
 202    return load_length

 219def build_cost_matrix(ref, hyp):
 220    """Build the cost (intersection) matrix between reference and hypothesis.
 221
 222    Args:
 223        ref: a list of tuples for the ground truth, where each tuple is
 224            (speaker, start, end) of type (string, float, float)
 225        hyp: a list of tuples for the diarization result hypothesis, same type
 226            as `ref`
 227
 228    Returns:
 229        a 2-dim numpy array, whose element (i, j) is the overlap duration
 230        between `i`th reference speaker and `j`th hypothesis speaker
 231    """
 232    ref = merge_speaker_intervals(ref)
 233    hyp = merge_speaker_intervals(hyp)
 234    ref_index = build_speaker_index(ref)
 235    hyp_index = build_speaker_index(hyp)
 236    cost_matrix = np.zeros((len(ref_index), len(hyp_index)))
 237    for ref_element in ref:
 238        for hyp_element in hyp:
 239            i = ref_index[ref_element[0]]
 240            j = hyp_index[hyp_element[0]]
 241            cost_matrix[i, j] += compute_intersection_length(
 242                ref_element, hyp_element)
 243    return cost_matrix

 246def compute_merged_exclusion_intervals(ref, collar):
 247    """Compute merged exclusion intervals based on reference boundaries.
 248
 249    Args:
 250        ref: a list of tuples for the ground truth
 251        collar: float, no-score tolerance around each reference boundary
 252            ([boundary - collar, boundary + collar])
 253
 254    Returns:
 255        a list of (start, end) tuples, sorted and merged
 256    """
 257    if isinstance(collar, bool) or not isinstance(
 258            collar, (int, float, np.number)):
 259        raise TypeError("Collar must be a number.")
 260    if not np.isfinite(collar) or collar < 0.0:
 261        raise ValueError("Collar must be a non-negative finite number.")
 262    collar = float(collar)
 263    if collar == 0.0:
 264        return []
 265
 266    ref = merge_speaker_intervals(ref)
 267    intervals = []
 268    for element in ref:
 269        start, end = element[1], element[2]
 270        intervals.append((start - collar, start + collar))
 271        intervals.append((end - collar, end + collar))
 272
 273    return _merge_intervals(intervals)

 276def compute_overlap_exclusion_intervals(ref):
 277    """Compute merged intervals where multiple reference speakers are active.
 278
 279    Args:
 280        ref: a list of (speaker, start, end) tuples for the ground truth
 281
 282    Returns:
 283        a list of (start, end) tuples where N_ref(t) > 1, sorted and merged
 284    """
 285    ref = merge_speaker_intervals(ref)
 286    boundaries = set()
 287    for element in ref:
 288        boundaries.add(element[1])
 289        boundaries.add(element[2])

 292    overlap_intervals = []

 300        if ref_count > 1:

 306def subtract_intervals(segments, exclusions):
 307    """Subtract exclusion intervals from segments.
 308
 309    Args:
 310        segments: a list of (speaker, start, end)
 311        exclusions: a list of (start, end) tuples, sorted and merged
 312
 313    Returns:
 314        a list of (speaker, start, end) with exclusions removed
 315    """
 316    if not exclusions:
 317        return segments
 318
 319    new_segments = []
 320    for speaker, start, end in segments:
 321        current_time = start
 322        for ex_start, ex_end in exclusions:
 323            if ex_end <= current_time:
 324                continue
 325            if ex_start >= end:
 326                break
 327
 328            if ex_start > current_time:
 329                new_segments.append((speaker, current_time, ex_start))
 330
 331            current_time = max(current_time, ex_end)
 332
 333            if current_time >= end:
 334                break
 335
 336        if current_time < end:
 337            new_segments.append((speaker, current_time, end))
 338
 339    return new_segments

 342def _preprocess_segments(ref, hyp, collar=0.0, skip_overlap=False):
 343    """Validate and preprocess reference and hypothesis segments."""
 344    check_input(ref)
 345    check_input(hyp)
 346    if not isinstance(skip_overlap, bool):
 347        raise TypeError("skip_overlap must be a boolean.")
 348    if not isinstance(collar, bool) and isinstance(
 349            collar, (int, float, np.number)):
 350        if not np.isfinite(collar) or collar < 0.0:
 351            raise ValueError("Collar must be a non-negative finite number.")
 352    else:
 353        raise TypeError("Collar must be a number.")
 354
 355    ref = merge_speaker_intervals(ref)
 356    hyp = merge_speaker_intervals(hyp)
 357
 358    exclusions = []
 359    if collar > 0.0:
 360        exclusions.extend(compute_merged_exclusion_intervals(ref, collar))
 361    if skip_overlap:
 362        exclusions.extend(compute_overlap_exclusion_intervals(ref))
 363    if exclusions:
 364        exclusions = _merge_intervals(exclusions)
 365        ref = subtract_intervals(ref, exclusions)
 366        hyp = subtract_intervals(hyp, exclusions)
 367
 368    return ref, hyp

 371def DER(ref, hyp, collar=0.0, skip_overlap=False, detailed=False):
 372    """Compute Diarization Error Rate (DER).
 373
 374    Args:
 375        ref: a list of tuples for the ground truth, where each tuple is
 376            (speaker, start, end) of type (string, float, float)
 377        hyp: a list of tuples for the diarization result hypothesis, same type
 378            as `ref`
 379        collar: float, no-score zone (in seconds) around each reference
 380            speaker boundary ([boundary - collar, boundary + collar])
 381        skip_overlap: bool, if True, exclude regions where more than one
 382            reference speaker is active
 383        detailed: bool, if True, return a dict containing DER, error component
 384            durations (miss, false_alarm, confusion, ref_length), and the
 385            optimal speaker_mapping
 386
 387    Returns:
 388        If `detailed=False`, a float number for the Diarization Error Rate.
 389        If `detailed=True`, a dict with keys:
 390            - "der": float, Diarization Error Rate
 391            - "miss": float, missed speech duration in seconds
 392            - "false_alarm": float, false alarm duration in seconds
 393            - "confusion": float, speaker confusion duration in seconds
 394            - "ref_length": float, scored reference length in seconds
 395            - "speaker_mapping": dict mapping matched ref -> hyp speaker
 396    """
 397    if not isinstance(detailed, bool):
 398        raise TypeError("detailed must be a boolean.")
 399    ref, hyp = _preprocess_segments(
 400        ref, hyp, collar=collar, skip_overlap=skip_overlap)
 401
 402    ref_total_length = compute_total_length(ref)
 403    ref_index = build_speaker_index(ref)
 404    hyp_index = build_speaker_index(hyp)
 405    cost_matrix = build_cost_matrix(ref, hyp)
 406    row_index, col_index = optimize.linear_sum_assignment(-cost_matrix)
 407    optimal_match_overlap = float(cost_matrix[row_index, col_index].sum())
 408
 409    inv_ref = {i: spk for spk, i in ref_index.items()}
 410    inv_hyp = {j: spk for spk, j in hyp_index.items()}
 411    speaker_mapping = {
 412        inv_ref[r]: inv_hyp[c]
 413        for r, c in zip(row_index, col_index)
 414        if cost_matrix[r, c] > 0.0
 415    }
 416
 417    miss, false_alarm, min_overlap, load_length = compute_error_components(
 418        ref, hyp)
 419    confusion = max(0.0, min_overlap - optimal_match_overlap)
 420
 421    if ref_total_length == 0.0:
 422        der_val = 0.0
 423    else:
 424        der_val = (load_length - optimal_match_overlap) / ref_total_length
 425
 426    if detailed:
 427        return {
 428            "der": float(der_val),
 429            "miss": float(miss),
 430            "false_alarm": float(false_alarm),
 431            "confusion": float(confusion),
 432            "ref_length": float(ref_total_length),
 433            "speaker_mapping": speaker_mapping,
 434        }
 435    return float(der_val)

 438def JER(ref, hyp, collar=0.0, skip_overlap=False, detailed=False):
 439    """Compute Jaccard Error Rate (JER).
 440
 441    JER is the average over all reference speakers of (1 - Jaccard similarity)
 442    under an optimal 1-to-1 mapping between reference and hypothesis speakers:
 443
 444        JER_r = 1 - |r \\cap h| / |r \\cup h|
 445        JER = (1 / |S_ref|) * \\sum_{r \\in S_ref} JER_r
 446
 447    Unmapped reference speakers have JER_r = 1.0.
 448
 449    Args:
 450        ref: a list of tuples for the ground truth, where each tuple is
 451            (speaker, start, end) of type (string, float, float)
 452        hyp: a list of tuples for the diarization result hypothesis, same type
 453            as `ref`
 454        collar: float, no-score zone (in seconds) around each reference
 455            speaker boundary ([boundary - collar, boundary + collar])
 456        skip_overlap: bool, if True, exclude regions where more than one
 457            reference speaker is active
 458        detailed: bool, if True, return a dict containing overall JER,
 459            per-reference-speaker JER, and the optimal speaker_mapping
 460
 461    Returns:
 462        If `detailed=False`, a float number for the Jaccard Error Rate.
 463        If `detailed=True`, a dict with keys:
 464            - "jer": float, Jaccard Error Rate
 465            - "speaker_jer": dict mapping each ref speaker -> speaker JER
 466            - "speaker_mapping": dict mapping matched ref -> hyp speaker
 467    """
 468    if not isinstance(detailed, bool):
 469        raise TypeError("detailed must be a boolean.")
 470    ref, hyp = _preprocess_segments(
 471        ref, hyp, collar=collar, skip_overlap=skip_overlap)
 472
 473    ref_index = build_speaker_index(ref)
 474    hyp_index = build_speaker_index(hyp)
 475
 476    if not ref_index:
 477        if detailed:
 478            return {
 479                "jer": 0.0,
 480                "speaker_jer": {},
 481                "speaker_mapping": {},
 482            }
 483        return 0.0
 484
 485    ref_durations = np.zeros(len(ref_index))
 486    for spk, start, end in ref:
 487        ref_durations[ref_index[spk]] += end - start
 488
 489    hyp_durations = np.zeros(len(hyp_index))
 490    for spk, start, end in hyp:
 491        hyp_durations[hyp_index[spk]] += end - start
 492
 493    intersection_matrix = build_cost_matrix(ref, hyp)
 494    union_matrix = (
 495        ref_durations[:, None] + hyp_durations[None, :] - intersection_matrix
 496    )
 497    jaccard_matrix = np.divide(
 498        intersection_matrix,
 499        union_matrix,
 500        out=np.zeros_like(intersection_matrix),
 501        where=union_matrix > 0.0,
 502    )
 503
 504    row_index, col_index = optimize.linear_sum_assignment(-jaccard_matrix)
 505
 506    inv_ref = {i: spk for spk, i in ref_index.items()}
 507    inv_hyp = {j: spk for spk, j in hyp_index.items()}
 508    speaker_jer = {spk: 1.0 for spk in ref_index}
 509    speaker_mapping = {}
 510
 511    for r, c in zip(row_index, col_index):
 512        ref_spk = inv_ref[r]
 513        speaker_jer[ref_spk] = float(1.0 - jaccard_matrix[r, c])
 514        if intersection_matrix[r, c] > 0.0:
 515            speaker_mapping[ref_spk] = inv_hyp[c]
 516
 517    jer_val = float(sum(speaker_jer.values()) / len(ref_index))
 518
 519    if detailed:
 520        return {
 521            "jer": jer_val,
 522            "speaker_jer": speaker_jer,
 523            "speaker_mapping": speaker_mapping,
 524        }
 525    return jer_val

 553def normalize_text(text):
 554    """Normalize text by lowercasing and stripping punctuation.
 555
 556    Follows the normalization convention in Google's DiarizationLM: converts
 557    to lowercase and removes punctuation characters while preserving token
 558    boundaries.
 559
 560    Args:
 561        text: input string
 562
 563    Returns:
 564        normalized string
 565    """
 566    text_lower = text.lower().strip()
 567    words = text_lower.split()
 568    new_words = []
 569    for word in words:
 570        new_word = word
 571        for punc in PUNCTUATIONS:
 572            replaced = new_word.replace(punc, "")
 573            if len(replaced.split()) != 1:
 574                continue
 575            new_word = replaced
 576        new_words.append(new_word)
 577    return " ".join(new_words)

 590def levenshtein_with_edits(ref_words, hyp_words):
 591    """Compute word-level Levenshtein distance and alignment.
 592
 593    Args:
 594        ref_words: a list of reference word strings
 595        hyp_words: a list of hypothesis word strings
 596
 597    Returns:
 598        a tuple (edit_distance, align), where `align` is a list of
 599        (ref_pos, hyp_pos) index pairs. Insertions have ref_pos == -1 and
 600        deletions have hyp_pos == -1.
 601    """
 602    n1 = len(ref_words)
 603    n2 = len(hyp_words)
 604    costs = [[0] * (n2 + 1) for _ in range(n1 + 1)]
 605    backptr = [[_CORRECT] * (n2 + 1) for _ in range(n1 + 1)]
 606
 607    for i in range(1, n1 + 1):
 608        costs[i][0] = i
 609        backptr[i][0] = _DELETION
 610
 611    for j in range(n2):
 612        costs[0][j + 1] = j + 1
 613        backptr[0][j + 1] = _INSERTION
 614        w2 = hyp_words[j]
 615        for i in range(n1):
 616            ins = costs[i + 1][j] + 1
 617            del_ = costs[i][j + 1] + 1
 618            same = (ref_words[i] == w2)
 619            sub = costs[i][j] + (0 if same else 1)
 620            best = min(ins, del_, sub)
 621            costs[i + 1][j + 1] = best
 622            if best == ins:
 623                backptr[i + 1][j + 1] = _INSERTION
 624            elif best == del_:
 625                backptr[i + 1][j + 1] = _DELETION
 626            elif same:
 627                backptr[i + 1][j + 1] = _CORRECT
 628            else:
 629                backptr[i + 1][j + 1] = _SUBSTITUTION
 630
 631    align = []
 632    i = n1
 633    j = n2
 634    while i > 0 or j > 0:
 635        ed_op = backptr[i][j]
 636        if ed_op == _INSERTION:
 637            align.append((-1, j - 1))
 638            j -= 1
 639        elif ed_op == _DELETION:
 640            align.append((i - 1, -1))
 641            i -= 1
 642        else:
 643            align.append((i - 1, j - 1))
 644            i -= 1

 651def WDER(ref, hyp, detailed=False):
 652    """Compute Word Diarization Error Rate (WDER).
 653
 654    WDER measures the fraction of aligned words (correct ASR words plus
 655    substituted ASR words after Levenshtein alignment between reference and
 656    hypothesis transcripts) that are assigned to the wrong speaker after
 657    optimal 1-to-1 speaker mapping (Shafey et al., Interspeech 2019).
 658
 659    Args:
 660        ref: a list of (speaker, word) tuples of type (str, str) for the
 661            ground truth. Each tuple's text may be a single word or
 662            whitespace-separated words spoken by that speaker.
 663        hyp: a list of (speaker, word) tuples of type (str, str) for the
 664            hypothesis, same format as `ref`.
 665        detailed: bool, if True, return a dict containing WDER, word counts
 666            (wder_sub, wder_correct, wder_total), WER breakdown, and the
 667            optimal speaker_mapping.
 668
 669    Returns:
 670        If `detailed=False`, a float number for the Word Diarization Error
 671        Rate.
 672        If `detailed=True`, a dict with keys:
 673            - "wder": float, Word Diarization Error Rate
 674            - "wder_sub": int, aligned words with wrong speaker label
 675            - "wder_correct": int, aligned words with correct speaker label
 676            - "wder_total": int, total aligned words (wer_correct + wer_sub)
 677            - "wer": float, Word Error Rate of the transcript
 678            - "wer_insert": int, inserted words
 679            - "wer_delete": int, deleted words
 680            - "wer_sub": int, substituted words
 681            - "wer_correct": int, correct words
 682            - "wer_total": int, total reference words
 683            - "speaker_mapping": dict mapping matched ref -> hyp speaker
 684    """
 685    if not isinstance(detailed, bool):
 686        raise TypeError("detailed must be a boolean.")
 687    check_word_input(ref)
 688    check_word_input(hyp)
 689
 690    ref_expanded = _expand_word_tuples(ref)
 691    hyp_expanded = _expand_word_tuples(hyp)
 692
 693    ref_spks = [spk for spk, _ in ref_expanded]
 694    ref_words = [w for _, w in ref_expanded]
 695    hyp_spks = [spk for spk, _ in hyp_expanded]
 696    hyp_words = [w for _, w in hyp_expanded]
 697
 698    _, align = levenshtein_with_edits(ref_words, hyp_words)
 699
 700    wer_insert = 0
 701    wer_delete = 0
 702    wer_sub = 0
 703    wer_correct = 0
 704    aligned_spk_pairs = []
 705
 706    for i, j in align:
 707        if i == -1:
 708            wer_insert += 1
 709        elif j == -1:
 710            wer_delete += 1
 711        else:
 712            if ref_words[i] == hyp_words[j]:
 713                wer_correct += 1
 714            else:
 715                wer_sub += 1
 716            aligned_spk_pairs.append((ref_spks[i], hyp_spks[j]))
 717
 718    wer_total = len(ref_words)
 719    wer_val = (
 720        0.0 if wer_total == 0
 721        else float((wer_insert + wer_delete + wer_sub) / wer_total)
 722    )
 723
 724    wder_total = len(aligned_spk_pairs)
 725    if wder_total == 0:
 726        wder_correct = 0
 727        wder_sub = 0
 728        wder_val = 0.0
 729        speaker_mapping = {}
 730    else:
 731        ref_index = build_speaker_index(ref_expanded)
 732        hyp_index = build_speaker_index(hyp_expanded)
 733        cost_matrix = np.zeros((len(ref_index), len(hyp_index)), dtype=int)
 734        for r_spk, h_spk in aligned_spk_pairs:
 735            cost_matrix[ref_index[r_spk], hyp_index[h_spk]] += 1
 736
 737        row_index, col_index = optimize.linear_sum_assignment(
 738            cost_matrix, maximize=True)
 739        wder_correct = int(cost_matrix[row_index, col_index].sum())
 740        wder_sub = wder_total - wder_correct
 741        wder_val = float(wder_sub / wder_total)
 742
 743        inv_ref = {i: spk for spk, i in ref_index.items()}
 744        inv_hyp = {j: spk for spk, j in hyp_index.items()}
 745        speaker_mapping = {
 746            inv_ref[r]: inv_hyp[c]
 747            for r, c in zip(row_index, col_index)
 748            if cost_matrix[r, c] > 0
 749        }
 750
 751    if detailed:
 752        return {
 753            "wder": wder_val,
 754            "wder_sub": wder_sub,
 755            "wder_correct": wder_correct,
 756            "wder_total": wder_total,
 757            "wer": wer_val,
 758            "wer_insert": wer_insert,
 759            "wer_delete": wer_delete,
 760            "wer_sub": wer_sub,
 761            "wer_correct": wer_correct,
 762            "wer_total": wer_total,
 763            "speaker_mapping": speaker_mapping,
 764        }
 765    return wder_val

 993def load_rttm(rttm_source):
 994    """Load diarization segments from a NIST RTTM file or line iterable.
 995
 996    Each `SPEAKER` line in an RTTM file has the format:
 997        SPEAKER <rec_id> <channel> <start> <duration> <NA> <NA> <spk> <NA> <NA>
 998
 999    Args:
1000        rttm_source: path to an RTTM file (str or os.PathLike) or an iterable
1001            of RTTM lines (such as an open file object)
1002
1003    Returns:
1004        a dict mapping `recording_id` (str) to a list of
1005        `(speaker, start, end)` tuples
1006    """
1007    if isinstance(rttm_source, (str, os.PathLike)):
1008        with open(rttm_source, "r", encoding="utf-8") as f:
1009            lines = f.readlines()
1010    elif hasattr(rttm_source, "__iter__") and not isinstance(
1011            rttm_source, (bytes, bytearray)):
1012        lines = list(rttm_source)
1013    else:
1014        raise TypeError(
1015            "rttm_source must be a file path or an iterable of lines.")
1016
1017    recordings = {}
1018    for line in lines:
1019        stripped = line.strip()
1020        if not stripped or stripped.startswith(("#", ";")):
1021            continue
1022        parts = stripped.split()
1023        if parts[0].upper() != "SPEAKER":
1024            continue
1025        if len(parts) < 8:
1026            raise ValueError(f"Invalid RTTM SPEAKER line: {stripped}")
1027        rec_id = parts[1]
1028        start = float(parts[3])
1029        duration = float(parts[4])
1030        if not np.isfinite(start) or not np.isfinite(duration) or duration < 0:
1031            raise ValueError(
1032                f"Invalid start or duration in RTTM line: {stripped}")
1033        end = start + duration
1034        speaker = parts[7]
1035        recordings.setdefault(rec_id, []).append((speaker, start, end))
1036
1037    return recordings

439def JER(ref, hyp, collar=0.0, skip_overlap=False, detailed=False):
440    """Compute Jaccard Error Rate (JER).
441
442    JER is the average over all reference speakers of (1 - Jaccard similarity)
443    under an optimal 1-to-1 mapping between reference and hypothesis speakers:
444
445        JER_r = 1 - |r \\cap h| / |r \\cup h|
446        JER = (1 / |S_ref|) * \\sum_{r \\in S_ref} JER_r
447
448    Unmapped reference speakers have JER_r = 1.0.
449
450    Args:
451        ref: a list of tuples for the ground truth, where each tuple is
452            (speaker, start, end) of type (string, float, float)
453        hyp: a list of tuples for the diarization result hypothesis, same type
454            as `ref`
455        collar: float, no-score zone (in seconds) around each reference
456            speaker boundary ([boundary - collar, boundary + collar])
457        skip_overlap: bool, if True, exclude regions where more than one
458            reference speaker is active
459        detailed: bool, if True, return a dict containing overall JER,
460            per-reference-speaker JER, and the optimal speaker_mapping
461
462    Returns:
463        If `detailed=False`, a float number for the Jaccard Error Rate.
464        If `detailed=True`, a dict with keys:
465            - "jer": float, Jaccard Error Rate
466            - "speaker_jer": dict mapping each ref speaker -> speaker JER
467            - "speaker_mapping": dict mapping matched ref -> hyp speaker
468    """
469    if not isinstance(detailed, bool):
470        raise TypeError("detailed must be a boolean.")
471    ref, hyp = _preprocess_segments(
472        ref, hyp, collar=collar, skip_overlap=skip_overlap)
473
474    ref_index = build_speaker_index(ref)
475    hyp_index = build_speaker_index(hyp)
476
477    if not ref_index:
478        if detailed:
479            return {
480                "jer": 0.0,
481                "speaker_jer": {},
482                "speaker_mapping": {},
483            }
484        return 0.0
485
486    ref_durations = np.zeros(len(ref_index))
487    for spk, start, end in ref:
488        ref_durations[ref_index[spk]] += end - start
489
490    hyp_durations = np.zeros(len(hyp_index))
491    for spk, start, end in hyp:
492        hyp_durations[hyp_index[spk]] += end - start
493
494    intersection_matrix = build_cost_matrix(ref, hyp)
495    union_matrix = (
496        ref_durations[:, None] + hyp_durations[None, :] - intersection_matrix
497    )
498    jaccard_matrix = np.divide(
499        intersection_matrix,
500        union_matrix,
501        out=np.zeros_like(intersection_matrix),
502        where=union_matrix > 0.0,
503    )
504
505    row_index, col_index = optimize.linear_sum_assignment(-jaccard_matrix)
506
507    inv_ref = {i: spk for spk, i in ref_index.items()}
508    inv_hyp = {j: spk for spk, j in hyp_index.items()}
509    speaker_jer = {spk: 1.0 for spk in ref_index}
510    speaker_mapping = {}
511
512    for r, c in zip(row_index, col_index):
513        ref_spk = inv_ref[r]
514        speaker_jer[ref_spk] = float(1.0 - jaccard_matrix[r, c])
515        if intersection_matrix[r, c] > 0.0:
516            speaker_mapping[ref_spk] = inv_hyp[c]
517
518    jer_val = float(sum(speaker_jer.values()) / len(ref_index))
519
520    if detailed:
521        return {
522            "jer": jer_val,
523            "speaker_jer": speaker_jer,
524            "speaker_mapping": speaker_mapping,
525        }
526    return jer_val

592    """Compute word-level Levenshtein distance and alignment.
593
594    Args:
595        ref_words: a list of reference word strings
596        hyp_words: a list of hypothesis word strings
597
598    Returns:
599        a tuple (edit_distance, align), where `align` is a list of
600        (ref_pos, hyp_pos) index pairs. Insertions have ref_pos == -1 and
601        deletions have hyp_pos == -1.
602    """
603    n1 = len(ref_words)
604    n2 = len(hyp_words)
605    costs = [[0] * (n2 + 1) for _ in range(n1 + 1)]
606    backptr = [[_CORRECT] * (n2 + 1) for _ in range(n1 + 1)]
607
608    for i in range(1, n1 + 1):
609        costs[i][0] = i
610        backptr[i][0] = _DELETION
611
612    for j in range(n2):
613        costs[0][j + 1] = j + 1
614        backptr[0][j + 1] = _INSERTION
615        w2 = hyp_words[j]
616        for i in range(n1):
617            ins = costs[i + 1][j] + 1
618            del_ = costs[i][j + 1] + 1
619            same = (ref_words[i] == w2)
620            sub = costs[i][j] + (0 if same else 1)
621            best = min(ins, del_, sub)
622            costs[i + 1][j + 1] = best
623            if best == ins:
624                backptr[i + 1][j + 1] = _INSERTION
625            elif best == del_:
626                backptr[i + 1][j + 1] = _DELETION
627            elif same:
628                backptr[i + 1][j + 1] = _CORRECT
629            else:
630                backptr[i + 1][j + 1] = _SUBSTITUTION
631
632    align = []
633    i = n1
634    j = n2
635    while i > 0 or j > 0:
636        ed_op = backptr[i][j]
637        if ed_op == _INSERTION:
638            align.append((-1, j - 1))
639            j -= 1
640        elif ed_op == _DELETION:
641            align.append((i - 1, -1))

646            j -= 1
